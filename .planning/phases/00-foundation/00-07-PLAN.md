def test_sc4_structlog_redacts_api_key(tmp_path: Path, monkeypatch: pytest.MonkeyPatch) -> None:
    """SC4: structlog emits JSON with API keys redacted on a sample event."""
    # Reset the configure-once guard so this test runs fresh.
    import ls_equity_fund.logging as logging_mod
    monkeypatch.setattr(logging_mod, "_CONFIGURED", False)
    import logging as stdlib_logging
    stdlib_logging.getLogger().handlers.clear()

    from ls_equity_fund.config import LoggingConfig
    from ls_equity_fund.logging import REDACTED_PLACEHOLDER, configure_logging

    cfg = LoggingConfig(
        level="INFO",
        log_dir=str(tmp_path),
        json_renderer_when_non_tty=True,
        redact_keys=[],
    )
    configure_logging(cfg)

    log = structlog.get_logger("sc4")
    log.info("api_call_sample", api_key="sk-ant-DO-NOT-LEAK-AAAA-BBBB", user="alice")

    # Force flush.
    for h in stdlib_logging.getLogger().handlers:
        h.flush()

    today = dt.datetime.now(dt.timezone.utc).strftime("%Y-%m-%d")
    log_file = tmp_path / f"{today}.jsonl"
    assert log_file.exists()
    text = log_file.read_text()

    assert "sk-ant-DO-NOT-LEAK-AAAA-BBBB" not in text, "API key leaked into JSON log output"
    assert REDACTED_PLACEHOLDER in text, "redaction placeholder not present"

    # Each line is a JSON object — verify the api_key field specifically.
    found_api_call = False
    for line in text.splitlines():
        try:
            obj = json.loads(line)
        except json.JSONDecodeError:
            continue
        if obj.get("event") == "api_call_sample":
            found_api_call = True
            assert obj.get("api_key") == REDACTED_PLACEHOLDER
            assert obj.get("user") == "alice"
    assert found_api_call, "api_call_sample event not found in JSONL output"
