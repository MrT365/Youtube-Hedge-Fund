class TestPhase1SC3Form4Codes:

        from ls_equity_fund.data.providers.edgar_provider import EdgarProvider
        from ls_equity_fund.data.filings import _insert_filing, _insert_insider

        provider = EdgarProvider(sec_user_agent="Test test@example.com")

        assert len(rows) == 1
        assert rows[0]["transaction_code"] == code

        # Persist filings_metadata + insider_transactions
        _insert_filing(conn_for, {
            "accession_number": filename, "ticker": rows[0]["ticker"], "cik": "0000320193",
            "form_type": "4", "filed_date": "2026-04-15",
            "period_of_report": None, "filepath": str(FIXTURES / filename),
            "content_hash": None,
        })
        rows[0]["filed_date"] = "2026-04-15"
        _insert_insider(conn_for, rows[0])

        persisted = conn_for.execute(
            "SELECT transaction_code FROM insider_transactions "
            "WHERE accession_number=?",
            (filename,),
        ).fetchone()
        assert persisted["transaction_code"] == code

    def test_db_check_constraint_rejects_unknown_code(self, conn_for):
        """Schema-level CP3 guard."""
        with pytest.raises(sqlite3.IntegrityError):
            conn_for.execute(
                "INSERT INTO insider_transactions (accession_number, line_no, ticker, "
                "transaction_code, transaction_date, filed_date) "
                "VALUES ('test', 1, 'AAPL', 'X', '2026-01-01', '2026-01-01')"
            )

    def test_cluster_buys_count_only_p_codes(self, conn_for):
        """CP3 binding — P only counts toward directional signal."""
        from ls_equity_fund.data.insider import detect_cluster_buys
        base = (
            "INSERT INTO insider_transactions (accession_number, line_no, ticker, "
