```python
"""SCORE-09 — Sector-percentile-rank utility (the load-bearing primitive).

Every Wave 2 factor module returns a long-format DataFrame
(columns: ticker, sub_factor, raw_value); this module turns those raw
values into 0–100 percentile ranks within GICS sector.

Plan-level decisions (locked across all factor plans):
- PD-1: scipy.stats.rankdata(method='average') — average rank for ties
- PD-2: NaN raw_value excluded from rank, remains NaN in percentile_rank
- PD-3: N-guard — N=0 → all NaN; N=1 → 50.0 (neutral); N≥2 → rank/N*100
"""
from __future__ import annotations
import numpy as np
import pandas as pd
from scipy.stats import rankdata

def percentile_rank_within(values: np.ndarray) -> np.ndarray:
    """0–100 percentile rank within a single cohort. NaN-safe. method='average'."""
    out = np.full(values.shape, np.nan, dtype=float)
    arr = np.asarray(values, dtype=float)
    mask = ~np.isnan(arr)
    n = int(mask.sum())
    if n == 0: return out
    if n == 1: out[mask] = 50.0; return out
    out[mask] = rankdata(arr[mask], method="average") / n * 100.0
    return out

def compute_sector_percentile_rank(df: pd.DataFrame) -> pd.DataFrame:
    """Group by `sector`, compute percentile_rank of `raw_value` within each.

    Input columns: ticker, sector, raw_value (others passed through unchanged).
    Output: input + `percentile_rank` (float, 0–100, NaN where raw_value NaN).
    Also adds `n_in_sector` (count of valid raw_values in the rank cohort).
    """
    out = df.copy()
    out["percentile_rank"] = np.nan
    out["n_in_sector"] = 0
    for sector, group in df.groupby("sector", dropna=False):
        valid_mask = group["raw_value"].notna()
        n = int(valid_mask.sum())
        out.loc[group.index, "n_in_sector"] = n
        if n == 0: continue
        if n == 1:
            out.loc[group[valid_mask].index, "percentile_rank"] = 50.0
            continue
        ranks = rankdata(group.loc[valid_mask, "raw_value"].values, method="average")
        out.loc[group[valid_mask].index, "percentile_rank"] = ranks / n * 100.0
    return out

__all__ = ["percentile_rank_within", "compute_sector_percentile_rank"]
```

Wraps in a single `with conn:` transaction (RESEARCH §Batch insert pattern).

from __future__ import annotations
import sqlite3, time
import pandas as pd
import structlog
log = structlog.get_logger(__name__)

_FACTOR_SCORES_COLS = ("ticker","score_date","factor","sub_factor","raw_value",
                       "percentile_rank","sector","n_in_sector","sufficient_history","computed_at")
_PARENT_COLS = ("ticker","score_date","factor","parent_score","sector",
                "n_subfactors_used","computed_at")

def write_factor_scores(conn: sqlite3.Connection, rows: pd.DataFrame) -> int:
    """INSERT OR REPLACE rows into factor_scores. Returns N rows written.
    `rows` is a DataFrame with the 10 columns listed in _FACTOR_SCORES_COLS
    (computed_at injected here if missing). Uses pandas → list[tuple] adapter
    that converts pd.NA / np.nan → None for SQL NULL fidelity.
    """
    if rows.empty: return 0
    df = rows.copy()
    if "computed_at" not in df.columns:
        df["computed_at"] = int(time.time())
    if "sufficient_history" not in df.columns:
        df["sufficient_history"] = 1
    df = df[list(_FACTOR_SCORES_COLS)]
    # pandas → SQL NULL (None) for NaN
    payload = [tuple(None if pd.isna(v) else v for v in row) for row in df.itertuples(index=False, name=None)]
    sql = (f"INSERT OR REPLACE INTO factor_scores ({', '.join(_FACTOR_SCORES_COLS)}) "
           f"VALUES ({', '.join('?' * len(_FACTOR_SCORES_COLS))})")
    with conn:
        conn.executemany(sql, payload)
    log.info("factor_scores_written", rows=len(payload))
    return len(payload)

def write_parent_scores(conn: sqlite3.Connection, rows: pd.DataFrame) -> int:
    """INSERT OR REPLACE rows into factor_scores_parent. Mirror of above."""
    if rows.empty: return 0
    df = rows.copy()
    if "computed_at" not in df.columns: df["computed_at"] = int(time.time())
    df = df[list(_PARENT_COLS)]
    payload = [tuple(None if pd.isna(v) else v for v in row) for row in df.itertuples(index=False, name=None)]
    sql = (f"INSERT OR REPLACE INTO factor_scores_parent ({', '.join(_PARENT_COLS)}) "
           f"VALUES ({', '.join('?' * len(_PARENT_COLS))})")
    with conn:
        conn.executemany(sql, payload)
    log.info("factor_scores_parent_written", rows=len(payload))
    return len(payload)

**3. Create `src/ls_equity_fund/factors/composer.py`** — registry + parent-score computer:

```python
"""SCORE-09 parent-score composition + factor registry dispatch.

The registry pattern lets Wave 2 plans run in parallel: each factor plan's
factors/<name>.py imports `register_factor` and decorates its `compute_<name>`
function. Until Wave 2 lands, the registry holds 8 placeholder callables that
raise NotImplementedError. Plan 02-10 reads `FACTOR_REGISTRY` at run-scoring
time, so adding a 9th factor in v2 is purely additive.
"""
from __future__ import annotations
import sqlite3
from collections.abc import Callable
from datetime import date
import pandas as pd
import structlog
log = structlog.get_logger(__name__)

# The 8 canonical factor names (locked — also enforced by CHECK in migration 0003).
FACTOR_NAMES: tuple[str, ...] = (
    "momentum", "value", "quality", "growth",
    "revisions", "short_interest", "insider", "institutional",
)

# A factor compute function takes (conn, asof, tickers) and returns a long-format
# DataFrame with columns: ticker, sub_factor, raw_value (sector + factor are
# attached by the orchestrator from universe).
FactorComputeFn = Callable[[sqlite3.Connection, date, list[str] | None], pd.DataFrame]

def _placeholder(name: str) -> FactorComputeFn:
    def _fn(conn: sqlite3.Connection, asof: date, tickers: list[str] | None) -> pd.DataFrame:
        raise NotImplementedError(
            f"factor '{name}' not yet registered — Wave 2 plan owns this. "
            "If you see this in production, a Wave 2 plan failed to import."
        )
    return _fn

FACTOR_REGISTRY: dict[str, FactorComputeFn] = {n: _placeholder(n) for n in FACTOR_NAMES}

def register_factor(name: str) -> Callable[[FactorComputeFn], FactorComputeFn]:
    """Decorator: replace the placeholder in FACTOR_REGISTRY with the real fn."""
    if name not in FACTOR_NAMES:
        raise ValueError(f"unknown factor name {name!r}; expected one of {FACTOR_NAMES}")
    def decorator(fn: FactorComputeFn) -> FactorComputeFn:
        FACTOR_REGISTRY[name] = fn
        log.debug("factor_registered", name=name, fn=fn.__qualname__)
        return fn
    return decorator

def compute_parent_factor_score(subfactors_df: pd.DataFrame) -> pd.DataFrame:
    """SCORE-09: parent_score = mean of non-NULL sub-factor percentile_ranks (PD-4).

    Input cols:  ticker, score_date, factor, sub_factor, percentile_rank, sector
    Output cols: ticker, score_date, factor, parent_score, sector, n_subfactors_used
    """
    if subfactors_df.empty:
        return pd.DataFrame(columns=[
            "ticker", "score_date", "factor", "parent_score", "sector", "n_subfactors_used"
        ])
    grouped = subfactors_df.groupby(
        ["ticker", "score_date", "factor", "sector"], dropna=False, as_index=False
    ).agg(
        parent_score=("percentile_rank", "mean"),       # pandas mean() skips NaN by default
        n_subfactors_used=("percentile_rank", lambda s: int(s.notna().sum())),
    )
    return grouped[["ticker", "score_date", "factor", "parent_score", "sector", "n_subfactors_used"]]

__all__ = ["FACTOR_NAMES", "FACTOR_REGISTRY", "register_factor",
           "compute_parent_factor_score", "FactorComputeFn"]
```

from datetime import date as date_type, datetime

from ls_equity_fund.factors.composer import FACTOR_NAMES

def run_scoring(
    config_path: Path = typer.Option(Path("config.yaml"), "--config", help="Path to config.yaml"),
    env_path: Path = typer.Option(Path(".env"), "--env", help="Path to .env"),
    asof: str | None = typer.Option(None, "--asof",
        help="Score date (YYYY-MM-DD); default = today UTC. Past dates supported for backfill."),
    ticker: str | None = typer.Option(None, "--ticker", help="Restrict to one ticker (debug)"),
    sector: str | None = typer.Option(None, "--sector", help="Restrict to one GICS sector"),
    factors: str | None = typer.Option(None, "--factors",
        help=f"Comma-separated subset of {FACTOR_NAMES} (default: all 8)"),
    dry_run: bool = typer.Option(False, "--dry-run", help="Compute but do not persist"),
) -> None:
    """Compute L2 factor scores: 8 factors × 27 sub-factors, sector-percentile rank,
    persist to factor_scores + factor_scores_parent."""

    # --- Step 1: locate config.yaml ---
    if not config_path.exists():
        typer.secho(f"ERROR: config.yaml not found at {config_path.resolve()}", fg=typer.colors.RED, err=True)
        raise typer.Exit(code=2)

    # --- Step 2: locate .env ---
    if not env_path.exists():
        typer.secho(f"ERROR: .env not found at {env_path.resolve()}", fg=typer.colors.RED, err=True)
        raise typer.Exit(code=3)

    # --- Step 3: load_config ---
    try:
        config, secrets = load_config(yaml_path=config_path, env_path=env_path)
    except typer.Exit:
        raise
    except Exception as e:
        typer.secho(f"ERROR: failed to load config: {e}", fg=typer.colors.RED, err=True)
        raise typer.Exit(code=4) from e

    # --- Step 4: configure logging ---
    configure_logging(config.logging)
    log = structlog.get_logger("run-scoring")

    # --- Step 5: parse + validate flags ---
    asof_date: date_type = date_type.today() if asof is None else datetime.strptime(asof, "%Y-%m-%d").date()
    factor_names = [f.strip() for f in factors.split(",")] if factors else list(FACTOR_NAMES)
    unknown = [f for f in factor_names if f not in FACTOR_NAMES]
    if unknown:
        typer.secho(f"ERROR: unknown --factors values: {unknown}; expected subset of {FACTOR_NAMES}",
                    fg=typer.colors.RED, err=True)
        raise typer.Exit(code=5)

    log.info("run_scoring_invoked", asof=asof_date.isoformat(), ticker=ticker, sector=sector,
             factors=factor_names, dry_run=dry_run)

    # --- Step 6: SKELETON — orchestrator pending Plan 02-10 ---
    typer.secho(
        "run-scoring: orchestrator pending Plan 02-10. Skeleton only. "
        "Re-run after Plan 02-10 ships.",
        fg=typer.colors.YELLOW, err=True,
    )
    raise typer.Exit(code=97)


__all__ = ["run_scoring"]
