```python
"""SCORE-06 — Short Interest factor (3 sub-factors, side-aware via long-side semantics).

Per RESEARCH.md §6 + Plan-level decision: long-side semantics persisted; the
optimizer derives short-side score = 100 - long_side_score at read time.

Sub-factors (3) — all sign-flipped to long-side ("higher = better for longs"):
  short_pct_float_inv = -short_percent_of_float[asof]
  short_dtc_inv       = -short_ratio[asof]                            (short_ratio IS days-to-cover)
  short_change_inv    = -(short_percent_of_float[asof] - short_percent_of_float[asof-30d])
                      = SI[asof-30d] - SI[asof]                       (declining SI = bullish for longs → positive)

Pitfall 6 binding: every sub-factor name carries `_inv` suffix so the
sign-convention audit trail is in the name itself.
"""
from __future__ import annotations
import sqlite3
from datetime import date, timedelta
import numpy as np
import pandas as pd
import structlog

from ls_equity_fund.factors.composer import register_factor

log = structlog.get_logger(__name__)

SUB_FACTORS: tuple[str, ...] = ("short_pct_float_inv", "short_dtc_inv", "short_change_inv")

def _latest_si_at_or_before(
    conn: sqlite3.Connection, ticker: str, target: date,
) -> dict[str, float | None] | None:
    cur = conn.execute(
        "SELECT short_percent_of_float, short_ratio "
        "FROM short_interest WHERE ticker=? AND snapshot_date <= ? "
        "ORDER BY snapshot_date DESC LIMIT 1",
        (ticker, target.isoformat()),
    )
    row = cur.fetchone()
    if not row: return None
    return {
        "short_percent_of_float": float(row[0]) if row[0] is not None else None,
        "short_ratio": float(row[1]) if row[1] is not None else None,
    }

@register_factor("short_interest")
def compute_short(
    conn: sqlite3.Connection, asof: date, tickers: list[str] | None,
) -> pd.DataFrame:
    if tickers:
        placeholders = ",".join("?" * len(tickers))
        univ = [r[0] for r in conn.execute(
            f"SELECT ticker FROM universe WHERE ticker IN ({placeholders})", tickers).fetchall()]
    else:
        univ = [r[0] for r in conn.execute("SELECT ticker FROM universe").fetchall()]
    if not univ:
        return pd.DataFrame(columns=["ticker","sub_factor","raw_value"])

    rows: list[dict] = []
    asof_minus_30 = asof - timedelta(days=30)

    for ticker in univ:
        latest = _latest_si_at_or_before(conn, ticker, asof)
        prior_30 = _latest_si_at_or_before(conn, ticker, asof_minus_30)

        if latest is None:
            short_pct_float_inv = np.nan
            short_dtc_inv = np.nan
            short_change_inv = np.nan
        else:
            spf = latest.get("short_percent_of_float")
            sr  = latest.get("short_ratio")
            short_pct_float_inv = -float(spf) if spf is not None else np.nan
            short_dtc_inv       = -float(sr)  if sr  is not None else np.nan

            if prior_30 is None or prior_30.get("short_percent_of_float") is None or spf is None:
                short_change_inv = np.nan
            else:
                # Declining SI is bullish for longs → invert the change so positive = better
                short_change_inv = -(float(spf) - float(prior_30["short_percent_of_float"]))

        for sub, val in [
            ("short_pct_float_inv", short_pct_float_inv),
            ("short_dtc_inv", short_dtc_inv),
            ("short_change_inv", short_change_inv),
        ]:
            rows.append({"ticker": ticker, "sub_factor": sub, "raw_value": val})

    out = pd.DataFrame(rows, columns=["ticker","sub_factor","raw_value"])
    log.info("compute_short_complete", n_tickers=len(univ), n_rows=len(out))
    return out

__all__ = ["compute_short", "SUB_FACTORS"]
```
