```python
"""Piotroski F-Score implementation (RESEARCH §3 §Quality).

Sums 9 binary checks across (current_year, prior_year) annual fundamentals.
Returns int ∈ {0,...,9} or None if prior_year is None or critical fields missing.

The 9 checks (verbatim from RESEARCH.md §3):
  Profitability (4):
    F1: net_income > 0
    F2: cfo > 0
    F3: (NI/TA)_t > (NI/TA)_t-1
    F4: cfo > net_income
  Leverage/Liquidity (3):
    F5: (LTD/TA)_t < (LTD/TA)_t-1
    F6: (CA/CL)_t > (CA/CL)_t-1
    F7: shares_outstanding_t ≤ shares_outstanding_t-1
  Efficiency (2):
    F8: (GP/Rev)_t > (GP/Rev)_t-1
    F9: (Rev/TA)_t > (Rev/TA)_t-1
"""
from __future__ import annotations
from typing import Any

PIOTROSKI_CHECKS: tuple[str, ...] = ("F1","F2","F3","F4","F5","F6","F7","F8","F9")

def _g(d: dict[str, Any], k: str) -> float | None:
    v = d.get(k)
    if v is None: return None
    try: return float(v)
    except (TypeError, ValueError): return None

def compute_piotroski_f(current: dict[str, Any], prior: dict[str, Any] | None) -> int | None:
    """Return 0-9 (sum of binary checks) or None if prior missing or critical field missing."""
    if prior is None: return None

    # Required fields (None on either side → return None for audit honesty)
    required_current = ("net_income","cfo","total_assets","long_term_debt",
                        "current_assets","current_liabilities","shares_outstanding",
                        "gross_profit","revenue")
    required_prior = required_current  # same fields needed
    for f in required_current:
        if _g(current, f) is None: return None
    for f in required_prior:
        if _g(prior, f) is None: return None

    ni_t  = _g(current, "net_income"); ni_p = _g(prior, "net_income")
    cfo_t = _g(current, "cfo")
    ta_t  = _g(current, "total_assets"); ta_p = _g(prior, "total_assets")
    ltd_t = _g(current, "long_term_debt"); ltd_p = _g(prior, "long_term_debt")
    ca_t  = _g(current, "current_assets"); ca_p = _g(prior, "current_assets")
    cl_t  = _g(current, "current_liabilities"); cl_p = _g(prior, "current_liabilities")
    sh_t  = _g(current, "shares_outstanding"); sh_p = _g(prior, "shares_outstanding")
    gp_t  = _g(current, "gross_profit"); gp_p = _g(prior, "gross_profit")
    rev_t = _g(current, "revenue"); rev_p = _g(prior, "revenue")

    # Guard against zero denominators (treat as failed check, not None — only NULL inputs trigger None)
    def _safe_div(n, d): return (n / d) if (d not in (0.0, None)) else None

    f1 = 1 if ni_t > 0 else 0
    f2 = 1 if cfo_t > 0 else 0
    roa_t = _safe_div(ni_t, ta_t); roa_p = _safe_div(ni_p, ta_p)
    f3 = 1 if (roa_t is not None and roa_p is not None and roa_t > roa_p) else 0
    f4 = 1 if cfo_t > ni_t else 0
    lev_t = _safe_div(ltd_t, ta_t); lev_p = _safe_div(ltd_p, ta_p)
    f5 = 1 if (lev_t is not None and lev_p is not None and lev_t < lev_p) else 0
    cur_t = _safe_div(ca_t, cl_t); cur_p = _safe_div(ca_p, cl_p)
    f6 = 1 if (cur_t is not None and cur_p is not None and cur_t > cur_p) else 0
    f7 = 1 if sh_t <= sh_p else 0
    gm_t = _safe_div(gp_t, rev_t); gm_p = _safe_div(gp_p, rev_p)
    f8 = 1 if (gm_t is not None and gm_p is not None and gm_t > gm_p) else 0
    at_t = _safe_div(rev_t, ta_t); at_p = _safe_div(rev_p, ta_p)
    f9 = 1 if (at_t is not None and at_p is not None and at_t > at_p) else 0

    return f1+f2+f3+f4+f5+f6+f7+f8+f9

__all__ = ["compute_piotroski_f", "PIOTROSKI_CHECKS"]
```

```python
"""Altman Z-Score implementation (RESEARCH §3 §Quality, Plan-level decision A3).

Original 1968 formula (manufacturing-firms model) — applied to all GICS sectors
per A3. The Z' (private) and Z'' (non-manufacturers) variants are NOT used in v1.

Z = 1.2*X1 + 1.4*X2 + 3.3*X3 + 0.6*X4 + 1.0*X5

  X1 = working_capital / total_assets
  X2 = retained_earnings / total_assets
  X3 = ebit / total_assets
  X4 = market_cap / total_liabilities
  X5 = revenue / total_assets

Zones:
  Z > 2.99    → 'safe'
  1.81 ≤ Z ≤ 2.99 → 'grey'
  Z < 1.81    → 'distress'
"""
from __future__ import annotations
from typing import Any

ALTMAN_COEFFS: tuple[float, float, float, float, float] = (1.2, 1.4, 3.3, 0.6, 1.0)

def compute_altman_z(fund: dict[str, Any], market_cap: float | None) -> float | None:
    """Return continuous Z score; None if denominators zero or critical inputs missing."""
    ta = fund.get("total_assets")
    if ta is None or ta == 0: return None
    tl = fund.get("total_liabilities")
    if tl is None or tl == 0: return None
    if market_cap is None: return None

    wc = fund.get("working_capital")
    re = fund.get("retained_earnings")
    ebit = fund.get("ebit")
    rev = fund.get("revenue")
    if any(v is None for v in (wc, re, ebit, rev)):
        return None

    try:
        x1 = float(wc) / float(ta)
        x2 = float(re) / float(ta)
        x3 = float(ebit) / float(ta)
        x4 = float(market_cap) / float(tl)
        x5 = float(rev) / float(ta)
    except (TypeError, ValueError, ZeroDivisionError):
        return None

    c1, c2, c3, c4, c5 = ALTMAN_COEFFS
    return c1*x1 + c2*x2 + c3*x3 + c4*x4 + c5*x5

def classify_zone(z: float | None) -> str | None:
    """Map Z to {'safe','grey','distress'}; None if z is None."""
    if z is None: return None
    if z > 2.99: return "safe"
    if z >= 1.81: return "grey"
    return "distress"

__all__ = ["compute_altman_z", "classify_zone", "ALTMAN_COEFFS"]
```

```python
"""SCORE-03 — Quality factor (8 sub-factors).

Per RESEARCH.md §3 Quality Cookbook + Plan-level decision A3 (original Altman Z).

Sub-factors (8):
  qual_roe_stability  = 1 / (std(roe[last 8 quarters]) + ε)   (higher = more stable)
  qual_gm_level       = fundamental_ratios.gross_margin       (passthrough)
  qual_gm_trend       = gm_t - gm_t-1y                         (annual fundamentals)
  qual_de_inv         = -fundamental_ratios.debt_to_equity    (sign-flipped, Pitfall 6)
  qual_cfo_ni         = fundamental_ratios.cfo_to_ni           (passthrough)
  qual_accruals_inv   = -fundamental_ratios.accruals_ratio     (sign-flipped, Pitfall 6)
  qual_piotroski      = 0–9 from _piotroski.compute_piotroski_f
  qual_altman_z       = continuous Z from _altman.compute_altman_z
"""
from __future__ import annotations
import sqlite3
from datetime import date
import numpy as np
import pandas as pd
import structlog

from ls_equity_fund.factors.composer import register_factor
from ls_equity_fund.factors._pit import latest_fundamentals_pit, latest_close_pit
from ls_equity_fund.factors._piotroski import compute_piotroski_f
from ls_equity_fund.factors._altman import compute_altman_z

log = structlog.get_logger(__name__)

SUB_FACTORS: tuple[str, ...] = (
    "qual_roe_stability", "qual_gm_level", "qual_gm_trend",
    "qual_de_inv", "qual_cfo_ni", "qual_accruals_inv",
    "qual_piotroski", "qual_altman_z",
)
_ROE_STABILITY_EPS = 1e-9   # avoid 1/0 when ROE is constant

@register_factor("quality")
def compute_quality(
    conn: sqlite3.Connection, asof: date, tickers: list[str] | None,
) -> pd.DataFrame:
    asof_iso = asof.isoformat()
    if tickers:
        placeholders = ",".join("?" * len(tickers))
        univ = [r[0] for r in conn.execute(
            f"SELECT ticker FROM universe WHERE ticker IN ({placeholders})", tickers).fetchall()]
    else:
        univ = [r[0] for r in conn.execute("SELECT ticker FROM universe").fetchall()]
    if not univ:
        return pd.DataFrame(columns=["ticker","sub_factor","raw_value"])

    # Pull fundamental_ratios (latest as_of_date <= asof) per ticker.
    placeholders = ",".join("?" * len(univ))
    ratios_q = f"""
        SELECT * FROM fundamental_ratios
        WHERE ticker IN ({placeholders})
        AND asof_date = (
            SELECT MAX(asof_date) FROM fundamental_ratios r2
            WHERE r2.ticker = fundamental_ratios.ticker AND r2.asof_date <= ?
        )
    """
    ratios_df = pd.read_sql_query(ratios_q, conn, params=[*univ, asof_iso])
    ratios_by_ticker = {r["ticker"]: r for _, r in ratios_df.iterrows()}

    # Pull last 8 quarterly fundamentals for ROE stability per ticker (separate loop).
    rows: list[dict] = []
    for ticker in univ:
        # 8 quarterly rows for ROE stability
        q_rows = latest_fundamentals_pit(conn, ticker, "quarterly", asof, n=8)
        roes: list[float] = []
        for q in q_rows:
            ni = q.get("net_income"); te = q.get("total_equity")
            if ni is not None and te not in (None, 0.0):
                try: roes.append(float(ni) / float(te))
                except (TypeError, ValueError): pass
        if len(roes) >= 8:
            std = float(np.std(roes, ddof=0))
            qual_roe_stability = 1.0 / (std + _ROE_STABILITY_EPS)
        else:
            qual_roe_stability = np.nan

        # 2 most recent ANNUAL rows for Piotroski + GM trend
        a_rows = latest_fundamentals_pit(conn, ticker, "annual", asof, n=2)
        current_annual = a_rows[0] if len(a_rows) >= 1 else {}
        prior_annual = a_rows[1] if len(a_rows) >= 2 else None

        # qual_gm_trend: GM_t - GM_t-1
        if current_annual and prior_annual:
            gp_t, rev_t = current_annual.get("gross_profit"), current_annual.get("revenue")
            gp_p, rev_p = prior_annual.get("gross_profit"), prior_annual.get("revenue")
            try:
                gm_t = float(gp_t)/float(rev_t) if rev_t not in (None,0.0) and gp_t is not None else None
                gm_p = float(gp_p)/float(rev_p) if rev_p not in (None,0.0) and gp_p is not None else None
                qual_gm_trend = (gm_t - gm_p) if (gm_t is not None and gm_p is not None) else np.nan
            except (TypeError, ValueError):
                qual_gm_trend = np.nan
        else:
            qual_gm_trend = np.nan

        # qual_piotroski (helper handles missing-prior → None)
        f_score = compute_piotroski_f(current_annual, prior_annual) if current_annual else None
        qual_piotroski = float(f_score) if f_score is not None else np.nan

        close = latest_close_pit(conn, ticker, asof)

        r = ratios_by_ticker.get(ticker, {})

        qual_gm_level   = float(gm) if gm is not None and not pd.isna(gm) else np.nan
        qual_de_inv     = -float(de) if de is not None and not pd.isna(de) else np.nan       # Pitfall 6
        qual_cfo_ni     = float(cn) if cn is not None and not pd.isna(cn) else np.nan
        qual_accruals_inv = -float(ac) if ac is not None and not pd.isna(ac) else np.nan      # Pitfall 6

        for sub, val in [
            ("qual_roe_stability", qual_roe_stability),
            ("qual_gm_level",      qual_gm_level),
            ("qual_gm_trend",      qual_gm_trend),
            ("qual_de_inv",        qual_de_inv),
            ("qual_cfo_ni",        qual_cfo_ni),
            ("qual_accruals_inv",  qual_accruals_inv),
            ("qual_piotroski",     qual_piotroski),
            ("qual_altman_z",      qual_altman_z),
        ]:
            rows.append({"ticker": ticker, "sub_factor": sub, "raw_value": val})

    out = pd.DataFrame(rows, columns=["ticker","sub_factor","raw_value"])
    log.info("compute_quality_complete", n_tickers=len(univ), n_rows=len(out))
    return out
```
