```python
"""SCORE-05 — Estimate Revisions factor (3 sub-factors).

Per RESEARCH.md §5 + Plan-level decision A4 (degenerate-neutral = 0.0 literal
zero, NOT NULL). Sufficient_history flag persisted alongside.

Sub-factors (3):
  rev_30d  = eps_fy1[asof] - eps_fy1[asof - 30 calendar days]
  rev_60d  = eps_fy1[asof] - eps_fy1[asof - 60 calendar days]
  rev_90d  = eps_fy1[asof] - eps_fy1[asof - 90 calendar days]

Calendar-day arithmetic (NOT trading days) is correct here because analyst
estimates are calendar-keyed snapshots. Pitfall 1 binding applies only to
momentum.

Pitfall 7 / A4 binding: when prior snapshot at the lookback window does not
exist (early in `analyst_estimates` lifetime), return raw_value=0.0 (literal
zero) so the ticker lands at the sector MEDIAN after percentile-rank, not
excluded from the cohort. Persist sufficient_history=0 as audit flag.

Returns DataFrame with FOUR columns: ticker, sub_factor, raw_value, sufficient_history.
"""
from __future__ import annotations
import sqlite3
from datetime import date, timedelta
import numpy as np
import pandas as pd
import structlog

from ls_equity_fund.factors.composer import register_factor

log = structlog.get_logger(__name__)

SUB_FACTORS: tuple[str, ...] = ("rev_30d", "rev_60d", "rev_90d")
LOOKBACKS: dict[str, int] = {"rev_30d": 30, "rev_60d": 60, "rev_90d": 90}

def _closest_snapshot_at_or_before(
    conn: sqlite3.Connection, ticker: str, target_date: date,
) -> float | None:
    cur = conn.execute(
        "SELECT eps_fy1 FROM analyst_estimates "
        "WHERE ticker=? AND snapshot_date <= ? "
        "ORDER BY snapshot_date DESC LIMIT 1",
        (ticker, target_date.isoformat()),
    )
    row = cur.fetchone()
    if not row or row[0] is None: return None
    try: return float(row[0])
    except (TypeError, ValueError): return None

@register_factor("revisions")
def compute_revisions(
    conn: sqlite3.Connection, asof: date, tickers: list[str] | None,
) -> pd.DataFrame:
    """Long-format DataFrame: (ticker, sub_factor, raw_value, sufficient_history).
    raw_value = eps_fy1[asof] - eps_fy1[asof-Nd]; degenerate-neutral=0.0 with
    sufficient_history=0 when prior snapshot missing.
    """
    if tickers:
        placeholders = ",".join("?" * len(tickers))
        univ = [r[0] for r in conn.execute(
            f"SELECT ticker FROM universe WHERE ticker IN ({placeholders})", tickers).fetchall()]
    else:
        univ = [r[0] for r in conn.execute("SELECT ticker FROM universe").fetchall()]
    if not univ:
        return pd.DataFrame(columns=["ticker","sub_factor","raw_value","sufficient_history"])

    rows: list[dict] = []
    for ticker in univ:
        # Latest snapshot (current) ≤ asof
        latest_eps = _closest_snapshot_at_or_before(conn, ticker, asof)

        for sub, lookback_days in LOOKBACKS.items():
            target = asof - timedelta(days=lookback_days)
            prior_eps = _closest_snapshot_at_or_before(conn, ticker, target)

            if latest_eps is None or prior_eps is None:
                # Degenerate-neutral (Pitfall 7 / A4)
                raw_value = 0.0
                sufficient_history = 0
            else:
                raw_value = latest_eps - prior_eps
                sufficient_history = 1

            rows.append({
                "ticker": ticker, "sub_factor": sub,
                "raw_value": raw_value, "sufficient_history": sufficient_history,
            })

    out = pd.DataFrame(rows, columns=["ticker","sub_factor","raw_value","sufficient_history"])
    log.info("compute_revisions_complete", n_tickers=len(univ), n_rows=len(out),
             n_degenerate=int((out["sufficient_history"] == 0).sum()))
    return out

__all__ = ["compute_revisions", "SUB_FACTORS"]
```
