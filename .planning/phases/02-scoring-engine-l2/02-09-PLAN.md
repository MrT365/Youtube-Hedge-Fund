Implement the Institutional Flow factor (SCORE-08) — 3 sub-factors derived from `institutional_holdings` (13F filings). """SCORE-08 — Institutional Flow factor (3 sub-factors).

```python
Per RESEARCH.md §8.

PIT correctness (D4 binding): latest period_end = MAX(period_end) restricted
by `filed_date <= asof`. Without this, a 13F filed Aug 14 for period Jun 30
would contaminate a July 1 backtest score.

Sub-factors (3):
  inst_fund_count            = COUNT(DISTINCT cik) at the latest known period_end
  inst_net_change            = SUM(change_shares × value_per_share) at latest period_end
                                value_per_share = value_usd / shares (where shares > 0)
  inst_multi_fund_open_flag  = 1 if 3+ tracked funds opened a new position in last 90 calendar days, else 0
"""
from __future__ import annotations
import sqlite3
from datetime import date, timedelta
import numpy as np
import pandas as pd
import structlog

from ls_equity_fund.factors.composer import register_factor

log = structlog.get_logger(__name__)

SUB_FACTORS: tuple[str, ...] = (
    "inst_fund_count", "inst_net_change", "inst_multi_fund_open_flag",
)
NEW_POSITION_WINDOW_DAYS = 90
MULTI_FUND_OPEN_THRESHOLD = 3


def _latest_period_end_pit(
    conn: sqlite3.Connection, ticker: str, asof: date,
) -> str | None:
    cur = conn.execute(
        "SELECT MAX(period_end) FROM institutional_holdings "
        "WHERE ticker=? AND filed_date <= ?",
        (ticker, asof.isoformat()),
    )
    row = cur.fetchone()
    if not row or row[0] is None: return None
    return str(row[0])


def _fund_count_at_period(
    conn: sqlite3.Connection, ticker: str, period_end: str, asof: date,
) -> int:
    cur = conn.execute(
        "SELECT COUNT(DISTINCT cik) FROM institutional_holdings "
        "WHERE ticker=? AND period_end=? AND filed_date <= ?",
        (ticker, period_end, asof.isoformat()),
    )
    return int(cur.fetchone()[0] or 0)


def _net_change_at_period(
    conn: sqlite3.Connection, ticker: str, period_end: str, asof: date,
) -> float | None:
    cur = conn.execute(
        "SELECT change_shares, value_usd, shares FROM institutional_holdings "
        "WHERE ticker=? AND period_end=? AND filed_date <= ?",
        (ticker, period_end, asof.isoformat()),
    )
    rows = cur.fetchall()
    if not rows: return None
    total = 0.0
    counted = 0
    for change_shares, value_usd, shares in rows:
        if change_shares is None or value_usd is None or shares is None: continue
        try:
            cs = float(change_shares); v = float(value_usd); s = float(shares)
        except (TypeError, ValueError): continue
        if s == 0.0: continue
        vps = v / s
        total += cs * vps
        counted += 1
    return total if counted > 0 else None


def _multi_fund_open_flag(
    conn: sqlite3.Connection, ticker: str, asof: date,
) -> int:
    start = (asof - timedelta(days=NEW_POSITION_WINDOW_DAYS)).isoformat()
    end = asof.isoformat()
    cur = conn.execute(
        "SELECT COUNT(DISTINCT cik) FROM institutional_holdings "
        "WHERE ticker=? AND is_new_position=1 AND filed_date BETWEEN ? AND ?",
        (ticker, start, end),
    )
    n = int(cur.fetchone()[0] or 0)
    return 1 if n >= MULTI_FUND_OPEN_THRESHOLD else 0


@register_factor("institutional")
def compute_institutional(
    conn: sqlite3.Connection, asof: date, tickers: list[str] | None,
) -> pd.DataFrame:
    if tickers:
        placeholders = ",".join("?" * len(tickers))
        univ = [r[0] for r in conn.execute(
            f"SELECT ticker FROM universe WHERE ticker IN ({placeholders})", tickers).fetchall()]
    else:
        univ = [r[0] for r in conn.execute("SELECT ticker FROM universe").fetchall()]
    if not univ:
        return pd.DataFrame(columns=["ticker","sub_factor","raw_value"])

    rows: list[dict] = []
    for ticker in univ:
        period_end = _latest_period_end_pit(conn, ticker, asof)
        if period_end is None:
            inst_fund_count = np.nan
            inst_net_change = np.nan
        else:
            count = _fund_count_at_period(conn, ticker, period_end, asof)
            inst_fund_count = float(count) if count > 0 else np.nan
            net = _net_change_at_period(conn, ticker, period_end, asof)
            inst_net_change = float(net) if net is not None else np.nan

        # multi_fund_open_flag is computed independently — needs 90d window, not latest period_end
        flag = _multi_fund_open_flag(conn, ticker, asof)
        # If there are no holdings at all for this ticker, treat the flag as NaN too (preserves audit completeness)
        if period_end is None and flag == 0:
            inst_multi = np.nan
        else:
            inst_multi = float(flag)

        for sub, val in [
            ("inst_fund_count", inst_fund_count),
            ("inst_net_change", inst_net_change),
            ("inst_multi_fund_open_flag", inst_multi),
        ]:
            rows.append({"ticker": ticker, "sub_factor": sub, "raw_value": val})

    out = pd.DataFrame(rows, columns=["ticker","sub_factor","raw_value"])
    log.info("compute_institutional_complete", n_tickers=len(univ), n_rows=len(out))
    return out


__all__ = ["compute_institutional", "SUB_FACTORS",
           "NEW_POSITION_WINDOW_DAYS", "MULTI_FUND_OPEN_THRESHOLD"]
```
