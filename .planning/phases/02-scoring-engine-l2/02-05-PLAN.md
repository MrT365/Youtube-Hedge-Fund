```python
"""SCORE-04 — Growth factor (5 sub-factors).

Per RESEARCH.md §4.

Sub-factors (5):
  grow_rev_yoy        = fundamental_ratios.revenue_growth_yoy        (passthrough)
  grow_earn_yoy       = fundamental_ratios.earnings_growth_yoy       (passthrough)
  grow_rev_accel      = revenue_growth_yoy[t] - revenue_growth_yoy[t-1y]
                          (derived; needs prior-year ratios snapshot)
  grow_rd_intensity   = fundamental_ratios.rd_intensity              (passthrough)
  grow_fcf_yoy        = (fcf_t - fcf_t-4q) / abs(fcf_t-4q)
                          (derived; uses 5 quarterly fundamentals rows or
                           2 annual rows — we use quarterly here for granularity)
"""
from __future__ import annotations
import sqlite3
from datetime import date, timedelta
import numpy as np
import pandas as pd
import structlog

from ls_equity_fund.factors.composer import register_factor
from ls_equity_fund.factors._pit import latest_fundamentals_pit

log = structlog.get_logger(__name__)

SUB_FACTORS: tuple[str, ...] = (
    "grow_rev_yoy", "grow_earn_yoy", "grow_rev_accel",
    "grow_rd_intensity", "grow_fcf_yoy",
)

@register_factor("growth")
def compute_growth(
    conn: sqlite3.Connection, asof: date, tickers: list[str] | None,
) -> pd.DataFrame:
    asof_iso = asof.isoformat()

    if tickers:
        placeholders = ",".join("?" * len(tickers))
        univ = [r[0] for r in conn.execute(
            f"SELECT ticker FROM universe WHERE ticker IN ({placeholders})", tickers).fetchall()]
    else:
        univ = [r[0] for r in conn.execute("SELECT ticker FROM universe").fetchall()]
    if not univ:
        return pd.DataFrame(columns=["ticker","sub_factor","raw_value"])

    placeholders = ",".join("?" * len(univ))

    # Latest ratios row at-or-before asof
    latest_ratios = pd.read_sql_query(
        f"""SELECT * FROM fundamental_ratios
            WHERE ticker IN ({placeholders})
            AND asof_date = (
                SELECT MAX(asof_date) FROM fundamental_ratios r2
                WHERE r2.ticker = fundamental_ratios.ticker AND r2.asof_date <= ?
            )""",
        conn, params=[*univ, asof_iso],
    )
    latest_by_ticker = {r["ticker"]: r for _, r in latest_ratios.iterrows()}

    # Prior-year ratios row: latest asof_date <= asof - 365 days
    # Use calendar timedelta here (NOT trading days) — annual ratios are
    # calendar-quarter-driven, not trading-day-driven. Pitfall 1 applies to
    # MOMENTUM not to ratios snapshots.
    prior_year_target = (asof - timedelta(days=365)).isoformat()
    prior_ratios = pd.read_sql_query(
        f"""SELECT * FROM fundamental_ratios
            WHERE ticker IN ({placeholders})
            AND asof_date = (
                SELECT MAX(asof_date) FROM fundamental_ratios r2
                WHERE r2.ticker = fundamental_ratios.ticker AND r2.asof_date <= ?
            )""",
        conn, params=[*univ, prior_year_target],
    )
    prior_by_ticker = {r["ticker"]: r for _, r in prior_ratios.iterrows()}

    rows: list[dict] = []
    for ticker in univ:
        cur_r = latest_by_ticker.get(ticker, {})
        prv_r = prior_by_ticker.get(ticker, {})

        rev_yoy_cur = cur_r.get("revenue_growth_yoy") if cur_r is not None else None
        earn_yoy_cur = cur_r.get("earnings_growth_yoy") if cur_r is not None else None
        rd_int = cur_r.get("rd_intensity") if cur_r is not None else None
        rev_yoy_prv = prv_r.get("revenue_growth_yoy") if prv_r is not None else None

        grow_rev_yoy  = float(rev_yoy_cur) if rev_yoy_cur is not None and not pd.isna(rev_yoy_cur) else np.nan
        grow_earn_yoy = float(earn_yoy_cur) if earn_yoy_cur is not None and not pd.isna(earn_yoy_cur) else np.nan
        grow_rd_int   = float(rd_int) if rd_int is not None and not pd.isna(rd_int) else np.nan

        if (rev_yoy_cur is not None and rev_yoy_prv is not None
                and not pd.isna(rev_yoy_cur) and not pd.isna(rev_yoy_prv)):
            grow_rev_accel = float(rev_yoy_cur) - float(rev_yoy_prv)
        else:
            grow_rev_accel = np.nan

        # FCF YoY: latest quarterly fcf / 4-quarters-ago fcf using PIT helper
        q_rows = latest_fundamentals_pit(conn, ticker, "quarterly", asof, n=5)
        fcf_t = None; fcf_p = None
        if len(q_rows) >= 1: fcf_t = q_rows[0].get("free_cash_flow")
        if len(q_rows) >= 5: fcf_p = q_rows[4].get("free_cash_flow")  # 4 quarters back
        if fcf_t is not None and fcf_p is not None:
            try:
                denom = abs(float(fcf_p))
                if denom == 0.0:
                    grow_fcf_yoy = np.nan
                else:
                    grow_fcf_yoy = (float(fcf_t) - float(fcf_p)) / denom
            except (TypeError, ValueError):
                grow_fcf_yoy = np.nan
        else:
            grow_fcf_yoy = np.nan

        for sub, val in [
            ("grow_rev_yoy", grow_rev_yoy),
            ("grow_earn_yoy", grow_earn_yoy),
            ("grow_rev_accel", grow_rev_accel),
            ("grow_rd_intensity", grow_rd_int),
            ("grow_fcf_yoy", grow_fcf_yoy),
        ]:
            rows.append({"ticker": ticker, "sub_factor": sub, "raw_value": val})

    out = pd.DataFrame(rows, columns=["ticker","sub_factor","raw_value"])
    log.info("compute_growth_complete", n_tickers=len(univ), n_rows=len(out))
    return out

__all__ = ["compute_growth", "SUB_FACTORS"]
```
