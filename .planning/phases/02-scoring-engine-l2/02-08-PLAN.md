```python
"""SCORE-07 — Insider Activity factor (3 sub-factors, CP3 P/S-only binding).

Per RESEARCH.md §7 + Plan-level decision PD-5.

CP3 BINDING (factor side):
  - ins_net_flow_90d: WHERE transaction_code IN ('P','S')
                      P contributes +total_value (bullish), S contributes -total_value (bearish)
                      A/M/F/G/D contribute ZERO (non-directional, excluded by filter)
  - ins_ceo_cfo_buys: WHERE transaction_code = 'P' (only P, since "buys" means buys)
                      AND is_officer=1
                      AND insider_title matches CEO_CFO_TITLE_RE (Python post-filter)
                      raw_value = 3 × SUM(total_value) (3× weight baked in)
  - ins_cluster_buy_count: WHERE transaction_code = 'P'
                           COUNT(DISTINCT insider_name) over 30-day window

Sector-median fallback (RESEARCH §7): when all 3 sub-factor raw values for a
ticker are NULL/NaN (no Form 4 activity in the window), replace with the
sector median computed from non-null tickers in the same GICS sector. This
keeps no-insider-data tickers at sector midpoint instead of dragging them
to the bottom of the rank. sufficient_history=0 audit flag set on those rows.

Returns DataFrame with FOUR columns: ticker, sub_factor, raw_value, sufficient_history.
"""
from __future__ import annotations
import sqlite3
from datetime import date, timedelta
import numpy as np
import pandas as pd
import structlog

from ls_equity_fund.factors.composer import register_factor
from ls_equity_fund.data.providers.edgar_provider import CEO_CFO_TITLE_RE

log = structlog.get_logger(__name__)

SUB_FACTORS: tuple[str, ...] = ("ins_net_flow_90d", "ins_ceo_cfo_buys", "ins_cluster_buy_count")
NET_FLOW_WINDOW_DAYS = 90
CLUSTER_WINDOW_DAYS = 30
CEO_CFO_WEIGHT = 3.0   # spec: 3× weight on CEO/CFO open-market purchases


def _net_flow_per_ticker(
    conn: sqlite3.Connection, asof: date, tickers: list[str],
) -> dict[str, float]:
    """SUM(P) - SUM(S) over the 90d window. CP3-bound: WHERE transaction_code IN ('P','S')."""
    start = (asof - timedelta(days=NET_FLOW_WINDOW_DAYS)).isoformat()
    end = asof.isoformat()
    placeholders = ",".join("?" * len(tickers))
    sql = (
        "SELECT ticker, "
        "SUM(CASE WHEN transaction_code='P' THEN total_value "
        "         WHEN transaction_code='S' THEN -total_value "
        "         ELSE 0 END) AS net_flow "
        "FROM insider_transactions "
        f"WHERE ticker IN ({placeholders}) "
        "  AND transaction_code IN ('P','S') "                                        # CP3 BINDING
        "  AND transaction_date BETWEEN ? AND ? "
        "GROUP BY ticker"
    )
    rows = conn.execute(sql, [*tickers, start, end]).fetchall()
    return {r[0]: float(r[1]) for r in rows if r[1] is not None}


def _ceo_cfo_buys_per_ticker(
    conn: sqlite3.Connection, asof: date, tickers: list[str],
) -> dict[str, float]:
    """3× SUM(total_value) of P-only purchases by officers whose title matches CEO_CFO_TITLE_RE.

    SQL filters transaction_code='P' AND is_officer=1; Python post-filters by regex.
    """
    start = (asof - timedelta(days=NET_FLOW_WINDOW_DAYS)).isoformat()
    end = asof.isoformat()
    placeholders = ",".join("?" * len(tickers))
    sql = (
        "SELECT ticker, insider_title, total_value FROM insider_transactions "
        f"WHERE ticker IN ({placeholders}) "
        "  AND transaction_code = 'P' "                                                # CP3 BINDING (P only)
        "  AND is_officer = 1 "
        "  AND transaction_date BETWEEN ? AND ?"
    )
    out: dict[str, float] = {}
    for ticker, title, total_value in conn.execute(sql, [*tickers, start, end]).fetchall():
        if total_value is None: continue
        if title and CEO_CFO_TITLE_RE.search(title):
            out[ticker] = out.get(ticker, 0.0) + float(total_value)
    # 3× weight applied at the end
    return {t: v * CEO_CFO_WEIGHT for t, v in out.items()}


def _cluster_buy_count_per_ticker(
    conn: sqlite3.Connection, asof: date, tickers: list[str],
) -> dict[str, int]:
    """COUNT(DISTINCT insider_name) of P-only purchases in the 30-day window.

    Mirrors data/insider.py::detect_cluster_buys — also CP3-bound (P only).
    """
    start = (asof - timedelta(days=CLUSTER_WINDOW_DAYS)).isoformat()
    end = asof.isoformat()
    placeholders = ",".join("?" * len(tickers))
    sql = (
        "SELECT ticker, COUNT(DISTINCT insider_name) AS distinct_buyers "
        "FROM insider_transactions "
        f"WHERE ticker IN ({placeholders}) "
        "  AND transaction_code = 'P' "                                                # CP3 BINDING (P only)
        "  AND transaction_date BETWEEN ? AND ? "
        "GROUP BY ticker"
    )
    rows = conn.execute(sql, [*tickers, start, end]).fetchall()
    return {r[0]: int(r[1]) for r in rows if r[1] is not None}


@register_factor("insider")
def compute_insider(
    conn: sqlite3.Connection, asof: date, tickers: list[str] | None,
) -> pd.DataFrame:
    """Compute 3 insider sub-factors with CP3 P/S-only binding + sector-median fallback."""
    if tickers:
        placeholders = ",".join("?" * len(tickers))
        univ_q = f"SELECT ticker, sector FROM universe WHERE ticker IN ({placeholders})"
        univ = pd.read_sql_query(univ_q, conn, params=tickers)
    else:
        univ = pd.read_sql_query("SELECT ticker, sector FROM universe", conn)
    if univ.empty:
        return pd.DataFrame(columns=["ticker","sub_factor","raw_value","sufficient_history"])

    target_tickers = univ["ticker"].tolist()
    sector_by_ticker = dict(zip(univ["ticker"], univ["sector"], strict=True))

    net_flow = _net_flow_per_ticker(conn, asof, target_tickers)
    ceo_cfo  = _ceo_cfo_buys_per_ticker(conn, asof, target_tickers)
    cluster  = _cluster_buy_count_per_ticker(conn, asof, target_tickers)

    # Build raw frame
    rows: list[dict] = []
    for ticker in target_tickers:
        raw_net = net_flow.get(ticker)
        raw_ceo = ceo_cfo.get(ticker)
        raw_clu = cluster.get(ticker)

        # If ALL 3 are missing, this ticker has no insider data at all → sector-median fallback path
        any_data = any(v is not None for v in (raw_net, raw_ceo, raw_clu))

        rows.append({"ticker": ticker, "sub_factor": "ins_net_flow_90d",
                     "raw_value": float(raw_net) if raw_net is not None else np.nan,
                     "sufficient_history": 1 if any_data else 0})
        rows.append({"ticker": ticker, "sub_factor": "ins_ceo_cfo_buys",
                     "raw_value": float(raw_ceo) if raw_ceo is not None else np.nan,
                     "sufficient_history": 1 if any_data else 0})
        rows.append({"ticker": ticker, "sub_factor": "ins_cluster_buy_count",
                     "raw_value": float(raw_clu) if raw_clu is not None else np.nan,
                     "sufficient_history": 1 if any_data else 0})

    df = pd.DataFrame(rows, columns=["ticker","sub_factor","raw_value","sufficient_history"])

    # Sector-median fallback for tickers with no insider data at all
    df["sector"] = df["ticker"].map(sector_by_ticker)
    for sub in SUB_FACTORS:
        sub_mask = df["sub_factor"] == sub
        for sector, group in df[sub_mask].groupby("sector"):
            valid = group[group["raw_value"].notna()]
            if valid.empty: continue
            median_val = float(valid["raw_value"].median())
            # Apply median to NaN rows in this (sub, sector) group whose sufficient_history=0
            fallback_idx = group[(group["raw_value"].isna()) & (group["sufficient_history"] == 0)].index
            df.loc[fallback_idx, "raw_value"] = median_val

    df = df.drop(columns=["sector"])
    log.info("compute_insider_complete", n_tickers=len(target_tickers), n_rows=len(df))
    return df


__all__ = ["compute_insider", "SUB_FACTORS",
           "NET_FLOW_WINDOW_DAYS", "CLUSTER_WINDOW_DAYS", "CEO_CFO_WEIGHT"]
```
