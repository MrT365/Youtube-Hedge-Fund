```python
"""SCORE-01 — Momentum factor (6 sub-factors).

Per Plan 02-02-PLAN.md and RESEARCH.md §1.

All offsets are TRADING-day offsets via pandas .shift(N), NOT calendar
timedelta math (Pitfall 1 binding).

Sub-factor table (6 emitted):
  mom_12_1       = close[asof-21bd] / close[asof-252bd] - 1     # Jegadeesh-Titman
  mom_6m         = close[asof]      / close[asof-126bd] - 1
  mom_3m         = close[asof]      / close[asof-63bd]  - 1
  mom_accel      = mom_3m - mom_6m
  mom_52w_high   = close[asof]      / max(close[asof-252bd:asof])
  mom_sector_rel = (close[asof]/close[asof-126bd]) / (etf[asof]/etf[asof-126bd])
                   where etf = sector_etfs[universe.sector[ticker]]
"""
from __future__ import annotations
import sqlite3
from datetime import date
import numpy as np
import pandas as pd
import structlog

from ls_equity_fund.factors.composer import register_factor
log = structlog.get_logger(__name__)

SUB_FACTORS: tuple[str, ...] = (
    "mom_12_1", "mom_6m", "mom_3m", "mom_accel", "mom_52w_high", "mom_sector_rel",
)

# Sector → ETF fallback mapping (used when config.data.sector_etfs not available).
# Standard GICS sector → SPDR sector ETF mapping per CLAUDE.md DATA-02.
_DEFAULT_SECTOR_ETFS: dict[str, str] = {
    "Information Technology": "XLK", "Communication Services": "XLC",
    "Health Care": "XLV", "Energy": "XLE", "Financials": "XLF",
    "Industrials": "XLI", "Consumer Discretionary": "XLY",
    "Consumer Staples": "XLP", "Materials": "XLB",
    "Real Estate": "XLRE", "Utilities": "XLU",
}

@register_factor("momentum")
def compute_momentum(
    conn: sqlite3.Connection,
    asof: date,
    tickers: list[str] | None,
) -> pd.DataFrame:
    """Return long-format DataFrame: (ticker, sub_factor, raw_value) for the 6 momentum sub-factors.

    Trading-day arithmetic is enforced via pandas .shift(N) on per-ticker close-price series sorted by date.
    Tickers with < 252 trading days of history get NaN for sub-factors that require the full lookback.
    """
    asof_iso = asof.isoformat()

    # 1. Resolve target tickers (intersect with universe.sector lookup for sector_rel).
    sector_q = "SELECT ticker, sector FROM universe"
    if tickers:
        placeholders = ",".join("?" * len(tickers))
        sector_q += f" WHERE ticker IN ({placeholders})"
        sector_df = pd.read_sql_query(sector_q, conn, params=tickers)
    else:
        sector_df = pd.read_sql_query(sector_q, conn)
    if sector_df.empty:
        return pd.DataFrame(columns=["ticker", "sub_factor", "raw_value"])

    target_tickers = sector_df["ticker"].tolist()

    # 2. Pull all closes for target tickers + relevant sector ETFs up to asof.
    sector_etfs_needed = sorted({_DEFAULT_SECTOR_ETFS.get(s) for s in sector_df["sector"]} - {None})
    all_tickers = sorted(set(target_tickers) | set(sector_etfs_needed))
    placeholders = ",".join("?" * len(all_tickers))
    prices_q = f"""
        SELECT ticker, date, close FROM daily_prices
        WHERE ticker IN ({placeholders}) AND date <= ?
        ORDER BY ticker, date
    """
    prices = pd.read_sql_query(prices_q, conn, params=[*all_tickers, asof_iso])
    if prices.empty:
        return pd.DataFrame(columns=["ticker", "sub_factor", "raw_value"])

    # 3. Per ticker: take the LAST 253 trading days (asof + 252 lookback). Compute sub-factors.
    rows: list[dict] = []

    # Build a {ticker: pd.Series of close indexed by date} mapping for fast .shift access.
    series_by_ticker: dict[str, pd.Series] = {}
    for tk, grp in prices.groupby("ticker"):
        grp_sorted = grp.sort_values("date")
        # .iloc[-253:] gives last 253 rows; positional indexing avoids any date arithmetic.
        s = grp_sorted["close"].reset_index(drop=True)
        series_by_ticker[tk] = s

    sector_to_etf = {row["ticker"]: _DEFAULT_SECTOR_ETFS.get(row["sector"])
                     for _, row in sector_df.iterrows()}

    for tk in target_tickers:
        s = series_by_ticker.get(tk)
        if s is None or len(s) == 0:
            for sub in SUB_FACTORS:
                rows.append({"ticker": tk, "sub_factor": sub, "raw_value": np.nan})
            continue

        n = len(s)
        # Position-based offsets — trading-day arithmetic via .iloc[-N]
        def _at_offset(series: pd.Series, n_back: int) -> float:
            """Return close N trading days back from the end. NaN if not enough history."""
            if len(series) <= n_back:
                return float("nan")
            return float(series.iloc[-1 - n_back])

        c0   = _at_offset(s, 0)        # close[asof]
        c21  = _at_offset(s, 21)       # close[asof-21bd]   (skip-1mo for 12-1)
        c63  = _at_offset(s, 63)       # close[asof-63bd]   (3mo)
        c126 = _at_offset(s, 126)      # close[asof-126bd]  (6mo)
        c252 = _at_offset(s, 252)      # close[asof-252bd]  (12mo)

        mom_12_1 = (c21 / c252 - 1.0) if (c252 not in (0.0, float("nan")) and not np.isnan(c21) and not np.isnan(c252)) else np.nan
        mom_6m   = (c0  / c126 - 1.0) if (c126 not in (0.0, float("nan")) and not np.isnan(c0)  and not np.isnan(c126)) else np.nan
        mom_3m   = (c0  / c63  - 1.0) if (c63  not in (0.0, float("nan")) and not np.isnan(c0)  and not np.isnan(c63))  else np.nan
        mom_accel = (mom_3m - mom_6m) if (not np.isnan(mom_3m) and not np.isnan(mom_6m)) else np.nan

        # 52w high: max over the last 252 trading days (inclusive of asof)
        if len(s) >= 252:
            window = s.iloc[-252:]
            hi = float(window.max())
            mom_52w_high = (c0 / hi) if (hi != 0.0 and not np.isnan(c0)) else np.nan
        else:
            mom_52w_high = np.nan

        # Sector-relative strength: stock 6m return / sector ETF 6m return
        etf_ticker = sector_to_etf.get(tk)
        mom_sector_rel = np.nan
        if etf_ticker and etf_ticker in series_by_ticker:
            etf_s = series_by_ticker[etf_ticker]
            etf_c0   = _at_offset(etf_s, 0)
            etf_c126 = _at_offset(etf_s, 126)
            if (not np.isnan(etf_c0) and not np.isnan(etf_c126) and etf_c126 != 0.0
                    and not np.isnan(c0) and not np.isnan(c126) and c126 != 0.0):
                stock_ret = c0 / c126
                etf_ret   = etf_c0 / etf_c126
                if etf_ret != 0.0:
                    mom_sector_rel = stock_ret / etf_ret

        rows.append({"ticker": tk, "sub_factor": "mom_12_1",       "raw_value": mom_12_1})
        rows.append({"ticker": tk, "sub_factor": "mom_6m",         "raw_value": mom_6m})
        rows.append({"ticker": tk, "sub_factor": "mom_3m",         "raw_value": mom_3m})
        rows.append({"ticker": tk, "sub_factor": "mom_accel",      "raw_value": mom_accel})
        rows.append({"ticker": tk, "sub_factor": "mom_52w_high",   "raw_value": mom_52w_high})
        rows.append({"ticker": tk, "sub_factor": "mom_sector_rel", "raw_value": mom_sector_rel})

    out = pd.DataFrame(rows, columns=["ticker", "sub_factor", "raw_value"])
    log.info("compute_momentum_complete", n_tickers=len(target_tickers), n_rows=len(out))
    return out

__all__ = ["compute_momentum", "SUB_FACTORS"]
```
