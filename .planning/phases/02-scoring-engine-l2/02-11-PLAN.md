class TestPhase2SC4PITReplay:
    def test_historical_replay_preserved(self, populated_db, cli_paths):
        # Run at asof=2026-04-01; snapshot factor_scores
        runner.invoke(app, ["run-scoring", "--config", str(cli_paths["config"]),
                            "--env", str(cli_paths["env"]), "--asof", "2026-04-01"])
        snapshot = populated_db.execute(
            "SELECT ticker, factor, sub_factor, raw_value, percentile_rank "
            "FROM factor_scores WHERE score_date='2026-04-01' ORDER BY ticker, factor, sub_factor"
        ).fetchall()
        # Run at asof=2026-05-04
        runner.invoke(app, ["run-scoring", "--config", str(cli_paths["config"]),
                            "--env", str(cli_paths["env"]), "--asof", "2026-05-04"])
        re_snapshot = populated_db.execute(
            "SELECT ticker, factor, sub_factor, raw_value, percentile_rank "
            "FROM factor_scores WHERE score_date='2026-04-01' ORDER BY ticker, factor, sub_factor"
        ).fetchall()
        assert snapshot == re_snapshot, "PIT replay violated: April 1 rows changed after May 4 run"

    def test_idempotent_same_day_rerun(self, populated_db, cli_paths):
        for _ in range(2):
            runner.invoke(app, ["run-scoring", "--config", str(cli_paths["config"]),
                                "--env", str(cli_paths["env"]), "--asof", "2026-05-04"])
        n = populated_db.execute(
            "SELECT COUNT(*) FROM factor_scores WHERE score_date='2026-05-04'"
        ).fetchone()[0]
        assert n == 30 * 27  # not duplicated

    def test_score_date_in_PK(self, populated_db):
        cols = populated_db.execute("PRAGMA table_info('factor_scores')").fetchall()
        pk_cols_in_order = sorted(
            [(c["pk"], c["name"]) for c in cols if c["pk"] > 0],
            key=lambda t: t[0],
        )
        names_in_order = [name for _, name in pk_cols_in_order]
        assert names_in_order == ["ticker", "score_date", "factor", "sub_factor"]
