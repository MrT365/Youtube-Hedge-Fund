```python
import time
import traceback
import uuid
from datetime import date as date_type
from typing import Any

from ls_equity_fund.config import Config
from ls_equity_fund.factors.persist import write_factor_scores, write_parent_scores
from ls_equity_fund.factors.sector_rank import compute_sector_percentile_rank
from ls_equity_fund.logging import bind_run_id

# Import side-effect: triggers @register_factor decorators on all 8 modules.
# This MUST be the canonical import point — without these imports, FACTOR_REGISTRY
# still holds placeholders that raise NotImplementedError.
from ls_equity_fund.factors import (  # noqa: F401  (import-for-side-effect)
    momentum, value, quality, growth, revisions, short, insider, institutional,
)


def run_scoring_pipeline(
    config: Config,
    conn: sqlite3.Connection,
    asof: date_type,
    *,
    ticker: str | None = None,
    sector: str | None = None,
    factors_subset: list[str] | None = None,
    dry_run: bool = False,
) -> dict[str, Any]:
    """Compose all factor modules into a single end-to-end scoring run.

    Steps:
      1. Open runs row (status='RUNNING'); bind run_id for structlog
      2. Enumerate scoring cohort from `universe` (PD-7: exclude delisted_date <= asof)
      3. Apply --ticker / --sector filters to cohort
      4. For each factor in factors_subset (default = all 8):
           df = FACTOR_REGISTRY[factor](conn, asof, cohort_tickers)
           Attach `factor` + `score_date` + `sector` columns
           df = compute_sector_percentile_rank(df)        # 0-100 within GICS sector
      5. Concatenate; compute parent_scores = compute_parent_factor_score(df)
      6. If not dry_run: write_factor_scores(conn, df); write_parent_scores(conn, parents)
      7. Close runs row (status='OK' / 'FAILED' + error)
    """
    run_id = str(uuid.uuid4())
    bind_run_id(run_id)
    start_ts = int(time.time())

    conn.execute(
        "INSERT INTO runs (run_id, start_ts, end_ts, status, error) "
        "VALUES (?, ?, NULL, 'RUNNING', NULL)",
        (run_id, start_ts),
    )

    factors_to_run = factors_subset or list(FACTOR_NAMES)
    manifest: dict[str, Any] = {
        "run_id": run_id, "asof": asof.isoformat(), "factors": factors_to_run,
        "dry_run": dry_run, "start_ts": start_ts,
    }
    overall_status = "OK"
    error_msg: str | None = None

    try:
        # 2. Cohort: live tickers as of asof (PD-7)
        cohort_q = (
            "SELECT ticker, sector FROM universe "
            "WHERE (delisted_date IS NULL OR delisted_date > ?)"
        )
        params: list[Any] = [asof.isoformat()]
        if ticker:
            cohort_q += " AND ticker = ?"; params.append(ticker)
        if sector:
            cohort_q += " AND sector = ?"; params.append(sector)
        cohort_df = pd.read_sql_query(cohort_q, conn, params=params)
        if cohort_df.empty:
            log.warning("scoring_pipeline_empty_cohort", asof=asof.isoformat())
            manifest["n_rows_factor_scores"] = 0
            manifest["n_rows_parent_scores"] = 0
            return _close_and_return(conn, run_id, manifest, overall_status, error_msg, start_ts)
        cohort_tickers = cohort_df["ticker"].tolist()
        sector_by_ticker = dict(zip(cohort_df["ticker"], cohort_df["sector"], strict=True))

        # 3. Run each factor; concatenate; rank within sector
        all_subfactor_rows: list[pd.DataFrame] = []
        for factor_name in factors_to_run:
            compute_fn = FACTOR_REGISTRY.get(factor_name)
            if compute_fn is None:
                log.warning("scoring_pipeline_unknown_factor", factor=factor_name)
                continue
            try:
                df = compute_fn(conn, asof, cohort_tickers)
            except Exception as e:
                log.error("factor_compute_failed", factor=factor_name, error=str(e))
                continue
            if df is None or df.empty: continue
            df = df.copy()
            df["factor"] = factor_name
            df["score_date"] = asof.isoformat()
            df["sector"] = df["ticker"].map(sector_by_ticker)
            # Rank: groupby (factor, sub_factor, sector) and compute percentile
            ranked_parts: list[pd.DataFrame] = []
            for (fac, sub), grp in df.groupby(["factor", "sub_factor"]):
                ranked = compute_sector_percentile_rank(grp)
                ranked_parts.append(ranked)
            df_ranked = pd.concat(ranked_parts, ignore_index=True) if ranked_parts else df
            # sufficient_history default = 1 if not present
            if "sufficient_history" not in df_ranked.columns:
                df_ranked["sufficient_history"] = 1
            all_subfactor_rows.append(df_ranked)

        if not all_subfactor_rows:
            manifest["n_rows_factor_scores"] = 0
            manifest["n_rows_parent_scores"] = 0
            return _close_and_return(conn, run_id, manifest, overall_status, error_msg, start_ts)

        all_df = pd.concat(all_subfactor_rows, ignore_index=True)

        # 4. Parent scores
        parents = compute_parent_factor_score(
            all_df[["ticker", "score_date", "factor", "sub_factor", "percentile_rank", "sector"]]
        )

        # 5. Persist (or skip if dry_run)
        if dry_run:
            log.info("scoring_pipeline_dry_run",
                     n_subfactor_rows=len(all_df), n_parent_rows=len(parents))
            manifest["n_rows_factor_scores"] = 0  # nothing written
            manifest["n_rows_parent_scores"] = 0
        else:
            write_factor_scores(conn, all_df)
            write_parent_scores(conn, parents)
            manifest["n_rows_factor_scores"] = len(all_df)
            manifest["n_rows_parent_scores"] = len(parents)

    except Exception as e:
        overall_status = "FAILED"
        error_msg = (str(e) + "\n" + traceback.format_exc())[:500]
        log.error("scoring_pipeline_failed", error=str(e))

    return _close_and_return(conn, run_id, manifest, overall_status, error_msg, start_ts)


def _close_and_return(
    conn: sqlite3.Connection, run_id: str, manifest: dict[str, Any],
    status: str, error: str | None, start_ts: int,
) -> dict[str, Any]:
    end_ts = int(time.time())
    conn.execute(
        "UPDATE runs SET end_ts=?, status=?, error=? WHERE run_id=?",
        (end_ts, status, error, run_id),
    )
    manifest["end_ts"] = end_ts
    manifest["duration_seconds"] = end_ts - start_ts
    manifest["status"] = status
    if error: manifest["error"] = error
    log.info("scoring_pipeline_complete",
             run_id=run_id, status=status, duration_seconds=end_ts - start_ts)
    return manifest

    "run_scoring_pipeline",
]
```

```python
"""``meridian run-scoring`` — Phase 2 L2 factor scoring CLI (Plan 02-10 fully wired).

Mirrors data_cmd.py's structure: locate config, load_config, configure_logging,
delegate to run_scoring_pipeline, print summary banner.

Exit code map:
    0  — success
    2  — config.yaml not found
    3  — .env not found
    4  — config validation failed
    5  — invalid --factors value (unknown factor name)
    6  — invalid --asof date format
    7  — unexpected pipeline failure
"""
from __future__ import annotations
from datetime import date as date_type, datetime
from pathlib import Path

import structlog
import typer

from ls_equity_fund.config import load_config
from ls_equity_fund.db import get_connection, get_db_path
from ls_equity_fund.factors.composer import FACTOR_NAMES, run_scoring_pipeline
from ls_equity_fund.logging import configure_logging


def run_scoring(
    config_path: Path = typer.Option(Path("config.yaml"), "--config"),
    env_path: Path = typer.Option(Path(".env"), "--env"),
    asof: str | None = typer.Option(None, "--asof",
        help="Score date (YYYY-MM-DD); default = today UTC."),
    ticker: str | None = typer.Option(None, "--ticker"),
    sector: str | None = typer.Option(None, "--sector"),
    factors: str | None = typer.Option(None, "--factors",
        help=f"Comma-separated subset of {FACTOR_NAMES} (default: all 8)"),
    dry_run: bool = typer.Option(False, "--dry-run"),
) -> None:
    """Compute L2 factor scores: 8 factors × 27 sub-factors, sector-percentile rank,
    persist to factor_scores."""

    if not config_path.exists():
        typer.secho(f"ERROR: config.yaml not found at {config_path.resolve()}",
                    fg=typer.colors.RED, err=True)
        raise typer.Exit(code=2)
    if not env_path.exists():
        typer.secho(f"ERROR: .env not found at {env_path.resolve()}",
                    fg=typer.colors.RED, err=True)
        raise typer.Exit(code=3)

    try:
        config, secrets = load_config(yaml_path=config_path, env_path=env_path)
    except typer.Exit:
        raise
    except Exception as e:
        typer.secho(f"ERROR: failed to load config: {e}",
                    fg=typer.colors.RED, err=True)
        raise typer.Exit(code=4) from e

    configure_logging(config.logging)
    log = structlog.get_logger("run-scoring")

    try:
        asof_date: date_type = (
            date_type.today() if asof is None
            else datetime.strptime(asof, "%Y-%m-%d").date()
        )
    except ValueError as e:
        typer.secho(f"ERROR: --asof must be YYYY-MM-DD, got {asof!r}: {e}",
                    fg=typer.colors.RED, err=True)
        raise typer.Exit(code=6) from e

    factor_names = [f.strip() for f in factors.split(",")] if factors else None
    if factor_names is not None:
        unknown = [f for f in factor_names if f not in FACTOR_NAMES]
        if unknown:
            typer.secho(
                f"ERROR: unknown --factors values: {unknown}; "
                f"expected subset of {list(FACTOR_NAMES)}",
                fg=typer.colors.RED, err=True,
            )
            raise typer.Exit(code=5)

    log.info("run_scoring_start", asof=asof_date.isoformat(), ticker=ticker,
             sector=sector, factors=factor_names, dry_run=dry_run)

    conn = get_connection(get_db_path(config))
    try:
        manifest = run_scoring_pipeline(
            config, conn, asof_date,
            ticker=ticker, sector=sector,
            factors_subset=factor_names, dry_run=dry_run,
        )
    except Exception as e:
        log.error("run_scoring_unexpected_failure", error=str(e))
        typer.secho(f"ERROR: pipeline failed: {e}",
                    fg=typer.colors.RED, err=True)
        raise typer.Exit(code=7) from e
    finally:
        conn.close()

    # Summary banner
    run_id_short = str(manifest.get("run_id", ""))[:8]
    duration = manifest.get("duration_seconds", 0)
    n_rows = manifest.get("n_rows_factor_scores", 0)
    n_parents = manifest.get("n_rows_parent_scores", 0)
    status = manifest.get("status", "?")
    color = typer.colors.GREEN if status == "OK" else typer.colors.YELLOW
    typer.secho(
```
