```python
"""PIT (point-in-time) helpers shared by Wave 2 factor modules.

Every fundamentals/estimates lookup MUST go through this module (Pitfall 8 binding —
ignoring `as_of_ingest_date` causes look-ahead contamination on backtest replay).

Functions:
  latest_fundamentals_pit(conn, ticker, period_type, asof, n=1) → list[dict]
      Latest n fundamentals rows for ticker AT-OR-BEFORE asof, with the
      latest-known restatement (max as_of_ingest_date). Mirrors
      data/ratios.py::_latest_per_period.

  latest_estimates_pit(conn, ticker, asof) → dict | None
      Latest analyst_estimates snapshot AT-OR-BEFORE asof.

  latest_close_pit(conn, ticker, asof) → float | None
      Latest daily_prices.close AT-OR-BEFORE asof.
"""
from __future__ import annotations
import sqlite3
from datetime import date
from typing import Any

_FUND_PIT_SQL = """
    WITH latest AS (
        SELECT ticker, period_end, period_type, MAX(as_of_ingest_date) AS aoid
        FROM fundamentals
        WHERE ticker = ? AND period_type = ?
          AND as_of_ingest_date <= ? AND period_end <= ?
        GROUP BY ticker, period_end, period_type
    )
    SELECT f.* FROM fundamentals f
    JOIN latest l USING (ticker, period_end, period_type)
    WHERE f.as_of_ingest_date = l.aoid
    ORDER BY f.period_end DESC
    LIMIT ?
"""

def latest_fundamentals_pit(
    conn: sqlite3.Connection,
    ticker: str,
    period_type: str,
    asof: date,
    n: int = 1,
) -> list[dict[str, Any]]:
    """Return the n most recent fundamentals rows for `ticker` at-or-before `asof`,
    each picking the latest-known restatement (max as_of_ingest_date).

    Empty list if no row matches.
    """
    cur = conn.execute(_FUND_PIT_SQL, (ticker, period_type, asof.isoformat(), asof.isoformat(), n))
    rows = cur.fetchall()
    if not rows: return []
    cols = [d[0] for d in cur.description]
    return [dict(zip(cols, r, strict=True)) for r in rows]

def latest_estimates_pit(
    conn: sqlite3.Connection, ticker: str, asof: date,
) -> dict[str, Any] | None:
    cur = conn.execute(
        "SELECT * FROM analyst_estimates "
        "WHERE ticker=? AND snapshot_date <= ? "
        "ORDER BY snapshot_date DESC LIMIT 1",
        (ticker, asof.isoformat()),
    )
    row = cur.fetchone()
    if not row: return None
    cols = [d[0] for d in cur.description]
    return dict(zip(cols, row, strict=True))

def latest_close_pit(
    conn: sqlite3.Connection, ticker: str, asof: date,
) -> float | None:
    cur = conn.execute(
        "SELECT close FROM daily_prices "
        "WHERE ticker=? AND date <= ? "
        "ORDER BY date DESC LIMIT 1",
        (ticker, asof.isoformat()),
    )
    row = cur.fetchone()
    return float(row[0]) if row and row[0] is not None else None

__all__ = ["latest_fundamentals_pit", "latest_estimates_pit", "latest_close_pit"]
```

```python
"""SCORE-02 — Value factor (6 sub-factors).

Per RESEARCH.md §2 Value Cookbook.

Plan-level decisions:
  - A1: EV/EBIT proxy (D&A not stored in L1 fundamentals); sub-factor is
        named val_ev_ebit_inv (NOT val_ev_ebitda_inv) — auditable name.
  - A2: total_debt = long_term_debt only (L1 lacks short-term debt).

Sub-factors (6):
  val_fwd_ey            = eps_fy1 / close                         (forward earnings yield)
  val_bp                = total_equity / market_cap               (book-to-price)
  val_fcf_yield         = fundamental_ratios.fcf_yield            (pass-through)
  val_ev_ebit_inv       = ebit / ev   (= 1 / (ev/ebit), inverted) (A1 proxy)
  val_shareholder_yield = dividend_yield + buyback_yield           (pass-through sum)
  val_sales_ev          = revenue / ev                            (sales-to-EV)

  where ev = market_cap + long_term_debt - cash_and_equivalents   (A2 simplification)
        market_cap = close × shares_outstanding   (close = latest close ≤ asof,
                                                   shares from latest fundamentals)
"""
from __future__ import annotations
import sqlite3
from datetime import date
import numpy as np
import pandas as pd
import structlog

from ls_equity_fund.factors.composer import register_factor
from ls_equity_fund.factors._pit import (
    latest_fundamentals_pit, latest_estimates_pit, latest_close_pit,
)
log = structlog.get_logger(__name__)

SUB_FACTORS: tuple[str, ...] = (
    "val_fwd_ey", "val_bp", "val_fcf_yield",
    "val_ev_ebit_inv", "val_shareholder_yield", "val_sales_ev",
)

@register_factor("value")
def compute_value(
    conn: sqlite3.Connection, asof: date, tickers: list[str] | None,
) -> pd.DataFrame:
    """Long-format DataFrame of (ticker, sub_factor, raw_value) for the 6 value sub-factors."""
    asof_iso = asof.isoformat()

    if tickers:
        placeholders = ",".join("?" * len(tickers))
        univ_q = f"SELECT ticker FROM universe WHERE ticker IN ({placeholders})"
        univ = [r[0] for r in conn.execute(univ_q, tickers).fetchall()]
    else:
        univ = [r[0] for r in conn.execute("SELECT ticker FROM universe").fetchall()]

    rows: list[dict] = []

    # Pull fundamental_ratios for everyone in one query (passthrough fields).
    if not univ:
        return pd.DataFrame(columns=["ticker", "sub_factor", "raw_value"])

    placeholders = ",".join("?" * len(univ))
    ratios_df = pd.read_sql_query(
        f"""SELECT * FROM fundamental_ratios
            WHERE ticker IN ({placeholders}) AND asof_date <= ?
            AND asof_date = (
                SELECT MAX(asof_date) FROM fundamental_ratios r2
                WHERE r2.ticker = fundamental_ratios.ticker AND r2.asof_date <= ?
            )""",
        conn, params=[*univ, asof_iso, asof_iso],
    )
    ratios_by_ticker = {r["ticker"]: r for _, r in ratios_df.iterrows()}

    for ticker in univ:
        # PIT-correct fundamentals (latest annual)
        fund_rows = latest_fundamentals_pit(conn, ticker, "annual", asof, n=1)
        fund = fund_rows[0] if fund_rows else {}

        # PIT-correct estimates + close
        est = latest_estimates_pit(conn, ticker, asof) or {}
        close = latest_close_pit(conn, ticker, asof)

        # Pass-through ratios
        r = ratios_by_ticker.get(ticker, {})
        fcf_yield = r.get("fcf_yield")
        dy = r.get("dividend_yield")
        by = r.get("buyback_yield")

        # Compute market cap
        shares = fund.get("shares_outstanding")
        market_cap = (close * shares) if (close is not None and shares is not None) else None

        # Compute EV (A2 simplification: total_debt = long_term_debt)
        ltd = fund.get("long_term_debt")
        cash = fund.get("cash_and_equivalents")
        ev = None
        if market_cap is not None and ltd is not None and cash is not None:
            ev = market_cap + ltd - cash

        # 1. val_fwd_ey
        eps_fy1 = est.get("eps_fy1")
        val_fwd_ey = (eps_fy1 / close) if (eps_fy1 is not None and close not in (None, 0.0)) else np.nan

        # 2. val_bp
        equity = fund.get("total_equity")
        val_bp = (equity / market_cap) if (equity is not None and market_cap not in (None, 0.0)) else np.nan

        # 3. val_fcf_yield (pass-through)
        val_fcf_yield = float(fcf_yield) if fcf_yield is not None and not pd.isna(fcf_yield) else np.nan

        # 4. val_ev_ebit_inv (A1 proxy: ebit/ev not ebitda/ev)
        ebit = fund.get("ebit")
        val_ev_ebit_inv = (ebit / ev) if (ebit is not None and ev not in (None, 0.0)) else np.nan

        # 5. val_shareholder_yield = dividend_yield + buyback_yield
        if dy is not None and by is not None and not pd.isna(dy) and not pd.isna(by):
            val_shareholder_yield = float(dy) + float(by)
        else:
            val_shareholder_yield = np.nan

        # 6. val_sales_ev
        revenue = fund.get("revenue")
        val_sales_ev = (revenue / ev) if (revenue is not None and ev not in (None, 0.0)) else np.nan

        for sub, val in [
            ("val_fwd_ey", val_fwd_ey),
            ("val_bp", val_bp),
            ("val_fcf_yield", val_fcf_yield),
            ("val_ev_ebit_inv", val_ev_ebit_inv),
            ("val_shareholder_yield", val_shareholder_yield),
            ("val_sales_ev", val_sales_ev),
        ]:
            rows.append({"ticker": ticker, "sub_factor": sub, "raw_value": val})

    out = pd.DataFrame(rows, columns=["ticker", "sub_factor", "raw_value"])
    log.info("compute_value_complete", n_tickers=len(univ), n_rows=len(out))
    return out

__all__ = ["compute_value", "SUB_FACTORS"]
```
